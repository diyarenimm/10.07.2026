            const symbols = ['💖', '✨', '💚', '🌟'];

            trail.style.position = 'fixed'; trail.style.left = x + 'px'; trail.style.top = y + 'px';

        function startSurprise() {
            const overlay = document.getElementById('fireworks-overlay');
            overlay.style.display = 'block';
            const symbols = ['🔥', '🎆', '🎇', '💥', '✨'];
            for(let i = 0; i < 20; i++) {
                const particle = document.createElement('div');
                particle.classList.add('explosion-particle');
                particle.innerHTML = symbols[Math.floor(Math.random() * symbols.length)];
                particle.style.left = (window.innerWidth / 2) + 'px';
                particle.style.top = (window.innerHeight / 2) + 'px';
                const angle = Math.random() * Math.PI * 2;
                const distance = Math.random() * 120 + 20;
                particle.style.setProperty('--tx', Math.cos(angle) * distance + 'px');
                particle.style.setProperty('--ty', Math.sin(angle) * distance + 'px');
                overlay.appendChild(particle);
            }
            const intro = document.getElementById('intro-screen');
            intro.style.opacity = '0';
            setTimeout(() => { intro.style.display = 'none'; overlay.style.display = 'none'; overlay.innerHTML = ''; }, 800);
            music.play().then(() => { musicBtn.innerText = "Durdur ⏸️"; }).catch(error => { console.log(error); });
        }

        const startDate = new Date("July 10, 2026 00:00:00").getTime();
        function updateTimer() {
            const now = new Date().getTime();
            const distance = now - startDate;
            if(distance > 0) {
                document.getElementById("days").innerText = Math.floor(distance / (1000 * 60 * 60 * 24));
                document.getElementById("hours").innerText = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
                document.getElementById("minutes").innerText = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
                document.getElementById("seconds").innerText = Math.floor((distance % (1000 * 60)) / 1000);
            }
        }
        setInterval(updateTimer, 1000);

        let lastIndices = {};
        function getSmartRandom(pool, elementId) {
            if(!lastIndices[elementId]) lastIndices[elementId] = -1;
            let idx;
            do { idx = Math.floor(Math.random() * pool.length); } while (pool.length > 1 && idx === lastIndices[elementId]);
            lastIndices[elementId] = idx;
            document.getElementById(elementId).innerText = pool[idx];
        }
